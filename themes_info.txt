--- JavaScript Switcher Function ---
function initThemePalette() {
  function applyPalette(paletteNum, persist = true) {
    ['theme-1', 'theme-2', 'theme-3', 'theme-4', 'theme-5', 'theme-6'].forEach((c, idx) => {
      if (String(idx + 1) === String(paletteNum)) {
        document.body.classList.add(c);
      } else {
        document.body.classList.remove(c);
      }
    });
    if (persist) localStorage.setItem('portfolio-theme-palette', String(paletteNum));
  }

  document.querySelectorAll('.theme-btn-option').forEach(btn => {
    btn.addEventListener('click', (e) => {
      e.preventDefault();
      applyPalette(btn.dataset.themeTarget, true);
    });
  });

  const saved = localStorage.getItem('portfolio-theme-palette');
  if (saved) applyPalette(saved, false);
}
